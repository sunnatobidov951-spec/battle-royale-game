// ===================== РЕНДЕР КАРТЫ ===================== const ctx = canvas.getContext('2d'); function resizeCanvas() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; } window.addEventListener('resize', resizeCanvas); resizeCanvas(); const WORLD_SCALE = 5; const MAP_OBJECTS = []; function seededRandom(seed) { let x = Math.sin(seed * 9999.91) * 43758.5453; return x - Math.floor(x); } // Генерируем стабильную карту for (let i = 0; i < 180; i++) { const x = 10 + seededRandom(i * 3 + 1) * (worldSize - 20); const z = 10 + seededRandom(i * 3 + 2) * (worldSize - 20); const type = i % 12 === 0 ? 'car' : i % 17 === 0 ? 'building' : i % 9 === 0 ? 'rock' : 'tree'; MAP_OBJECTS.push({ x, z, type, size: 0.8 + seededRandom(i * 7) * 1.2 }); } function worldToScreen(pos) { if (!myPlayer) { return { x: canvas.width / 2, y: canvas.height / 2 }; } const dx = (pos.x - myPlayer.position.x) * WORLD_SCALE; const dz = (pos.z - myPlayer.position.z) * WORLD_SCALE; return { x: canvas.width / 2 + dx, y: canvas.height / 2 + dz }; } function drawTree(x, y, s) { // тень ctx.fillStyle = 'rgba(0,0,0,0.25)'; ctx.beginPath(); ctx.ellipse(x, y + 9*s, 14*s, 7*s, 0, 0, Math.PI * 2); ctx.fill(); // ствол ctx.fillStyle = '#6b4226'; ctx.fillRect(x - 3*s, y, 6*s, 18*s); // крона ctx.fillStyle = '#174d27'; ctx.beginPath(); ctx.arc(x, y - 5*s, 15*s, 0, Math.PI * 2); ctx.fill(); ctx.fillStyle = '#26733a'; ctx.beginPath(); ctx.arc(x - 6*s, y - 10*s, 10*s, 0, Math.PI * 2); ctx.fill(); ctx.fillStyle = '#3d9149'; ctx.beginPath(); ctx.arc(x + 6*s, y - 8*s, 8*s, 0, Math.PI * 2); ctx.fill(); } function drawCar(x, y, s) { ctx.save(); ctx.translate(x, y); // тень ctx.fillStyle = 'rgba(0,0,0,0.3)'; ctx.fillRect(-22*s, -8*s, 44*s, 18*s); // кузов ctx.fillStyle = '#34495e'; ctx.fillRect(-20*s, -10*s, 40*s, 20*s); // кабина ctx.fillStyle = '#17202a'; ctx.fillRect(-10*s, -8*s, 20*s, 12*s); // окна ctx.fillStyle = '#8fd3ff'; ctx.fillRect(-8*s, -6*s, 7*s, 6*s); ctx.fillRect(1*s, -6*s, 7*s, 6*s); // колёса ctx.fillStyle = '#111'; ctx.beginPath(); ctx.arc(-14*s, 11*s, 5*s, 0, Math.PI * 2); ctx.arc(14*s, 11*s, 5*s, 0, Math.PI * 2); ctx.fill(); ctx.restore(); } function drawBuilding(x, y, s) { ctx.fillStyle = 'rgba(0,0,0,0.25)'; ctx.fillRect(x - 22*s, y - 18*s, 46*s, 40*s); ctx.fillStyle = '#8b7355'; ctx.fillRect(x - 20*s, y - 20*s, 40*s, 36*s); ctx.fillStyle = '#4b3024'; ctx.beginPath(); ctx.moveTo(x - 25*s, y - 20*s); ctx.lineTo(x, y - 35*s); ctx.lineTo(x + 25*s, y - 20*s); ctx.closePath(); ctx.fill(); ctx.fillStyle = '#222'; ctx.fillRect(x - 6*s, y - 5*s, 12*s, 21*s); ctx.fillStyle = '#b9e7ff'; ctx.fillRect(x - 16*s, y - 12*s, 8*s, 8*s); ctx.fillRect(x + 8*s, y - 12*s, 8*s, 8*s); } function drawRock(x, y, s) { ctx.fillStyle = 'rgba(0,0,0,0.25)'; ctx.beginPath(); ctx.ellipse(x, y + 6*s, 14*s, 7*s, 0, 0, Math.PI * 2); ctx.fill(); ctx.fillStyle = '#777'; ctx.beginPath(); ctx.moveTo(x - 13*s, y + 5*s); ctx.lineTo(x - 8*s, y - 10*s); ctx.lineTo(x + 5*s, y - 14*s); ctx.lineTo(x + 15*s, y); ctx.lineTo(x + 9*s, y + 10*s); ctx.closePath(); ctx.fill(); } function drawMapObject(obj) { const p = worldToScreen({ x: obj.x, z: obj.z }); if ( p.x < -100 || p.x > canvas.width + 100 || p.y < -100 || p.y > canvas.height + 100 ) return; if (obj.type === 'tree') { drawTree(p.x, p.y, obj.size); } else if (obj.type === 'car') { drawCar(p.x, p.y, obj.size); } else if (obj.type === 'building') { drawBuilding(p.x, p.y, obj.size); } else { drawRock(p.x, p.y, obj.size); } } function draw() { // земля ctx.fillStyle = '#4b6535'; ctx.fillRect(0, 0, canvas.width, canvas.height); // трава ctx.strokeStyle = 'rgba(120,150,70,0.35)'; ctx.lineWidth = 1; for (let x = 0; x < canvas.width; x += 35) { for (let y = 0; y < canvas.height; y += 35) { ctx.beginPath(); ctx.moveTo(x, y); ctx.lineTo(x + 4, y - 7); ctx.stroke(); } } if (latestState && myPlayer) { // объекты карты MAP_OBJECTS.forEach(drawMapObject); // зона const zonePos = worldToScreen({ x: latestState.zone.x, z: latestState.zone.z }); ctx.strokeStyle = 'rgba(80,170,255,0.95)'; ctx.lineWidth = 4; ctx.beginPath(); ctx.arc( zonePos.x, zonePos.y, latestState.zone.radius * WORLD_SCALE, 0, Math.PI * 2 ); ctx.stroke(); // лут latestState.loot.forEach(item => { const p = worldToScreen(item.position); if ( p.x > -20 && p.x < canvas.width + 20 && p.y > -20 && p.y < canvas.height + 20 ) { ctx.fillStyle = '#ffd54f'; ctx.beginPath(); ctx.arc(p.x, p.y, 6, 0, Math.PI * 2); ctx.fill(); ctx.strokeStyle = '#fff'; ctx.stroke(); } }); // боты (latestState.bots || []).forEach(bot => { if (!bot.alive) return; const p = worldToScreen(bot.position); ctx.fillStyle = '#e74c3c'; ctx.beginPath(); ctx.arc(p.x, p.y, 9, 0, Math.PI * 2); ctx.fill(); ctx.fillStyle = '#fff'; ctx.font = '10px Arial'; ctx.textAlign = 'center'; ctx.fillText('BOT', p.x, p.y - 13); }); // игроки latestState.players.forEach(pl => { if (!pl.alive) return; const p = worldToScreen(pl.position); ctx.fillStyle = pl.id === myId ? '#2ecc71' : '#3498db'; ctx.beginPath(); ctx.arc(p.x, p.y, 10, 0, Math.PI * 2); ctx.fill(); // направление const dirLen = 20; const dx = -Math.sin(pl.rotation.yaw) * dirLen; const dy = -Math.cos(pl.rotation.yaw) * dirLen; ctx.strokeStyle = '#fff'; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo(p.x, p.y); ctx.lineTo( p.x + dx, p.y + dy ); ctx.stroke(); }); // свой игрок const me = worldToScreen(myPlayer.position); ctx.strokeStyle = '#ffffff'; ctx.lineWidth = 3; ctx.beginPath(); ctx.arc(me.x, me.y, 15, 0, Math.PI * 2); ctx.stroke(); } requestAnimationFrame(draw); } draw();
