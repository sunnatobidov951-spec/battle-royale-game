} // ===================== ГЕНЕРАЦИЯ КАРТЫ ===================== for (let i = 0; i < 220; i++) { const x = 10 + seededRandom(i * 3 + 1) * (worldSize - 20); const z = 10 + seededRandom(i * 3 + 2) * (worldSize - 20); ctx.fill(); } // ===================== КАМЕНЬ ===================== function drawRock(x, y, s) { ctx.fillStyle = 'rgba(0,0,0,0.25)'; ctx.beginPath(); ctx.ellipse( x, y + 7 * s, 17 * s, 8 * s, 0, 0, Math.PI * 2 ); } // ===================== ОБЪЕКТЫ ===================== function drawMapObject(obj) { const p = worldToScreen({ x: obj.x, z: obj.z }); // не рисуем далеко за экраном if ( p.x < -150 || p.x > canvas.width + 150 || p.y < -150 || p.y > canvas.height + 150 ) { return; } } // ===================== ТРАВА ===================== function drawGround() { ctx.fillStyle = '#4b6535'; ctx.fillRect( 0, 0, canvas.width, canvas.height ); // мелкая трава ctx.strokeStyle = 'rgba(120,150,70,0.35)';
